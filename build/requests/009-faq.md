### site/tools/build.py: replace block_faq_groups
```python
def block_faq_groups():
    """FAQ: one list, four small group labels, each group a seated card with one drawn item on its edge. Answers may carry links, so they are written as markup in data/faq.json. The first row of each group starts open."""
    data = json.loads(read(DATA, 'faq.json'))
    out = []
    for g in data['groups']:
        rows = []
        for i, item in enumerate(g['items']):
            rows.append(
                '    <details class="m-faq"%s data-faq>\n'
                '      <summary class="m-faq__q"><h3 class="m-faq__title">%s</h3><span class="m-faq__plus" aria-hidden="true"></span></summary>\n'
                '      <div class="m-faq__a"><p class="m-faq__text">%s</p></div>\n'
                '    </details>' % (' open' if i == 0 else '', esc(item['q']), item['a']))
        art = g['art']
        out.append(
            '  <section class="m-seatCard o-faqList__group a-inview" id="%s" aria-labelledby="%s-title">\n'
            '    <div class="m-seatCard__art" aria-hidden="true"><img class="o-faqList__item -%s" src="assets/trucks/item-%s.svg" width="%d" height="%d" alt="" loading="lazy" decoding="async"></div>\n'
            '    <h2 class="o-faqList__label" id="%s-title">%s</h2>\n'
            '%s\n'
            '  </section>' % (esc(g['id']), esc(g['id']), esc(art['item']), esc(art['item']), art['w'], art['h'], esc(g['id']), esc(g['title']), '\n'.join(rows)))
    return '\n'.join(out)
```
